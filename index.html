import time import requests from datetime import datetime # ========================================== # 🤖 BEAST BOTH V6 - BUY ONLY # COINBASE MARKET DATA # PAPER TRADING ONLY # ========================================== STARTING_BALANCE = 20.00 STAKE = 1.00 MAX_POSITIONS = 2 TAKE_PROFIT_PERCENT = 0.80 STOP_LOSS_PERCENT = 1.50 CHECK_SECONDS = 15 PRODUCT = "BTC-USD" balance = STARTING_BALANCE positions = [] closed_trades = [] wins = 0 losses = 0 running = True BASE_URL = "https://api.exchange.coinbase.com" def get_price(): url = f"{BASE_URL}/products/{PRODUCT}/ticker" response = requests.get(url, timeout=10) response.raise_for_status() return float(response.json()["price"]) def get_candles(): url = f"{BASE_URL}/products/{PRODUCT}/candles" params = { "granularity": 300 } response = requests.get( url, params=params, timeout=10 ) response.raise_for_status() data = response.json() return sorted(data, key=lambda x: x[0]) def buy_signal(): candles = get_candles() if len(candles) < 30: return False closes = [float(c[4]) for c in candles] ema9 = sum(closes[-9:]) / 9 ema21 = sum(closes[-21:]) / 21 previous = closes[-2] current = closes[-1] # BUY ONLY - no sell-entry signals bullish = ( ema9 > ema21 and current > ema9 and current > previous ) return bullish def open_buy(price): global balance if len(positions) >= MAX_POSITIONS: return if balance < STAKE: return balance -= STAKE quantity = STAKE / price target = price * ( 1 + TAKE_PROFIT_PERCENT / 100 ) stop = price * ( 1 - STOP_LOSS_PERCENT / 100 ) positions.append({ "entry": price, "quantity": quantity, "stake": STAKE, "target": target, "stop": stop, "time": datetime.now().strftime("%H:%M:%S") }) print("\n🟢 BUY POSITION OPENED") print(f"Entry: ${price:.2f}") print(f"Take profit: ${target:.2f}") print(f"Stop loss: ${stop:.2f}") def close_position(index, price, reason): global balance, wins, losses position = positions.pop(index) proceeds = position["quantity"] * price profit = proceeds - position["stake"] balance += proceeds if profit > 0: wins += 1 else: losses += 1 closed_trades.append({ "profit": profit, "reason": reason, "price": price }) print("\n🔴 BUY POSITION CLOSED") print(f"Reason: {reason}") print(f"Exit: ${price:.2f}") print(f"Profit: ${profit:.4f}") print(f"Virtual balance: ${balance:.4f}") def manage_positions(price): for index in range(len(positions) - 1, -1, -1): position = positions[index] if price >= position["target"]: close_position( index, price, "TAKE PROFIT" ) elif price <= position["stop"]: close_position( index, price, "STOP LOSS" ) def dashboard(price): unrealized = sum( p["quantity"] * price - p["stake"] for p in positions ) equity = balance + sum( p["quantity"] * price for p in positions ) print("\n" + "=" * 48) print("🤖 BEAST BOTH V6 - BUY ONLY") print("COINBASE DATA | PAPER TRADING") print("=" * 48) print(f"💰 Virtual balance: ${balance:.4f}") print(f"📊 Total equity: ${equity:.4f}") print(f"📈 Realized P/L: ${balance - STARTING_BALANCE:.4f}") print(f"📉 Unrealized P/L: ${unrealized:.4f}") print(f"🟢 Wins: {wins} | 🔴 Losses: {losses}") print(f"📦 Closed trades: {len(closed_trades)}") print(f"📂 Open positions: {len(positions)}/{MAX_POSITIONS}") print(f"💵 {PRODUCT}: ${price:.2f}") for i, p in enumerate(positions, 1): print( f"BUY {i}: Entry ${p['entry']:.2f} | " f"TP ${p['target']:.2f} | " f"SL ${p['stop']:.2f}" ) print("=" * 48) def run(): global running print("🚀 BEAST BOTH V6 STARTED") print("BUY ONLY MODE") print("SELL ONLY AT TP OR PROTECTIVE STOP") print("PAPER TRADING ONLY") while running: try: price = get_price() # First manage existing BUY positions manage_positions(price) # Then search for new BUY opportunities if len(positions) < MAX_POSITIONS: if buy_signal(): open_buy(price) else: print("⏳ Waiting for BUY opportunity...") dashboard(price) time.sleep(CHECK_SECONDS) except KeyboardInterrupt: print("\n🛑 BOT STOPPED") break except Exception as error: print(f"⚠️ Error: {error}") time.sleep(CHECK_SECONDS) if __name__ == "__main__": run()
