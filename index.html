//+------------------------------------------------------------------+

//+------------------------------------------------------------------+
#property copyright "Combined Bot"
#property version   "2.10"
enum BOT_MODE { LITTLE = 0, FERRARI = 1 };
input BOT_MODE Mode = LITTLE;
input int EmaPeriod = 20;
input int MaxTradesPerDay = 4;
input ulong MagicNumber = 12345;
datetime lastDay = 0;
int tradesToday = 0;
int OnInit() { return(INIT_SUCCEEDED); }
void OnTick()
{
   double LotSize = 0.01;
   int SL_Points = 50;
   int TP_Points = 50;
   if(Mode == FERRARI) { LotSize = 0.05; SL_Points = 100; TP_Points = 200; }
   MqlDateTime dt;
   TimeToStruct(TimeCurrent(), dt);
   datetime today = StringToTime(IntegerToString(dt.year)+"."+IntegerToString(dt.mon)+"."+IntegerToString(dt.day));
   if(today != lastDay) { lastDay = today; tradesToday = 0; }
   if(tradesToday >= MaxTradesPerDay) return;
   if(PositionsTotal() > 0) return;
   double ema[];
   int handle = iMA(_Symbol, PERIOD_M5, EmaPeriod, 0, MODE_EMA, PRICE_CLOSE);
   CopyBuffer(handle, 0, 0, 2, ema);
   double price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   double emaValue = ema[0];
   double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
   double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
   MqlTradeRequest req; MqlTradeResult res; ZeroMemory(req);
   req.action = TRADE_ACTION_DEAL; req.symbol = _Symbol; req.volume = LotSize;
   req.magic = MagicNumber; req.type_filling = ORDER_FILLING_IOC;
   if(price > emaValue)
   {
      req.type = ORDER_TYPE_BUY; req.price = ask;
      req.sl = ask - SL_Points * point;
      req.tp = ask + TP_Points * point;
      if(OrderSend(req, res)) tradesToday++;
   }
   else if(price < emaValue)
   {
      req.type = ORDER_TYPE_SELL; req.price = bid;
      req.sl = bid + SL_Points * point;
      req.tp = bid - TP_Points * point;
      if(OrderSend(req, res)) tradesToday++;
   }
}
