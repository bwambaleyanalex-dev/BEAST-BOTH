//+------------------------------------------------------------------+ //| COMBINED BOT - Little Little + Ferrari - TEST ON DEMO FIRST | //+------------------------------------------------------------------+ #property copyright "Combined Bot" #property version "2.00" enum BOT_MODE { LITTLE = 0, FERRARI = 1 }; input BOT_MODE Mode = LITTLE; // Choose: LITTLE or FERRARI input int EmaPeriod = 20; // EMA for M5 input int MaxTradesPerDay = 4; // Stop after 4 trades input ulong MagicNumber = 12345; datetime lastDay = 0; int tradesToday = 0; int OnInit() { return(INIT_SUCCEEDED); } void OnTick() { // --- Settings per mode --- double LotSize = 0.01; int SL_Points = 50; // $0.50 int TP_Points = 50; // $0.50 if(Mode == FERRARI) { LotSize = 0.05; SL_Points = 100; // $1.00 = -$5.00 = ALL IN TP_Points = 200; // $2.00 = +$10.00 } // --- Reset each day --- MqlDateTime dt; TimeToStruct(TimeCurrent(), dt); datetime today = StringToTime(IntegerToString(dt.year)+"."+IntegerToString(dt.mon)+"."+IntegerToString(dt.day)); if(today != lastDay) { lastDay = today; tradesToday = 0; } if(tradesToday >= MaxTradesPerDay) return; if(PositionsTotal() > 0) return; // --- Get EMA 20 on M5 --- double ema[]; int handle = iMA(_Symbol, PERIOD_M5, EmaPeriod, 0, MODE_EMA, PRICE_CLOSE); CopyBuffer(handle, 0, 0, 2, ema); double price = SymbolInfoDouble(_Symbol, SYMBOL_BID); double emaValue = ema[0]; double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT); double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK); double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID); MqlTradeRequest req; MqlTradeResult res; ZeroMemory(req); req.action = TRADE_ACTION_DEAL; req.symbol = _Symbol; req.volume = LotSize; req.magic = MagicNumber; req.type_filling = ORDER_FILLING_IOC; // BUY if price ABOVE EMA if(price > emaValue) { req.type = ORDER_TYPE_BUY; req.price = ask; req.sl = ask - SL_Points * point; // FIXED - No *10 req.tp = ask + TP_Points * point; // FIXED - No *10 if(OrderSend(req, res)) tradesToday++; } // SELL if price BELOW EMA else if(price < emaValue) { req.type = ORDER_TYPE_SELL; req.price = bid; req.sl = bid + SL_Points * point; // FIXED req.tp = bid - TP_Points * point; // FIXED if(OrderSend(req, res)) tradesToday++; } }
