//+------------------------------------------------------------------+ //| Little Little Bot - Gradual Growth - TEST ON DEMO FIRST | //+------------------------------------------------------------------+ #property copyright "Little Little Bot" #property version "1.00" input double LotSize = 0.01; input int EmaPeriod = 20; input int StopLossPoints = 50; // $0.50 input int TakeProfitPoints = 50; // $0.50 input int MaxTradesPerDay = 4; input ulong MagicNumber = 12345; datetime lastDay = 0; int tradesToday = 0; //+------------------------------------------------------------------+ int OnInit() { return(INIT_SUCCEEDED); } //+------------------------------------------------------------------+ void OnTick() { // Reset counter each new day MqlDateTime dt; TimeToStruct(TimeCurrent(), dt); datetime today = StringToTime(IntegerToString(dt.year)+"."+IntegerToString(dt.mon)+"."+IntegerToString(dt.day)); if(today != lastDay) { lastDay = today; tradesToday = 0; } if(tradesToday >= MaxTradesPerDay) return; if(PositionsTotal() > 0) return; // Only 1 trade at a time // Get EMA 20 on M5 double ema[]; int handle = iMA(_Symbol, PERIOD_M5, EmaPeriod, 0, MODE_EMA, PRICE_CLOSE); CopyBuffer(handle, 0, 0, 2, ema); double price = SymbolInfoDouble(_Symbol, SYMBOL_BID); double emaValue = ema[0]; double sl = 0, tp = 0; double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT); double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK); double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID); MqlTradeRequest req; MqlTradeResult res; ZeroMemory(req); // BUY if price ABOVE EMA if(price > emaValue) { sl = ask - StopLossPoints * point * 10; tp = ask + TakeProfitPoints * point * 10; req.action = TRADE_ACTION_DEAL; req.symbol = _Symbol; req.volume = LotSize; req.type = ORDER_TYPE_BUY; req.price = ask; req.sl = sl; req.tp = tp; req.magic = MagicNumber; req.type_filling = ORDER_FILLING_IOC; if(OrderSend(req, res)) tradesToday++; } // SELL if price BELOW EMA else if(price < emaValue) { sl = bid + StopLossPoints * point * 10; tp = bid - TakeProfitPoints * point * 10; req.action = TRADE_ACTION_DEAL; req.symbol = _Symbol; req.volume = LotSize; req.type = ORDER_TYPE_SELL; req.price = bid; req.sl = sl; req.tp = tp; req.magic = MagicNumber; req.type_filling = ORDER_FILLING_IOC; if(OrderSend(req, res)) tradesToday++; } }
